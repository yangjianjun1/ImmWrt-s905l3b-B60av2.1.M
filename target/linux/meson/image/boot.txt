# S905L3B boot script - B60av2.1.M

setenv condev "console=ttyAML0,115200n8 no_console_suspend consoleblank=0"
setenv kernel_loadaddr "0x00208000"
setenv dtb_loadaddr "0x21800000"
setenv dtb_name "meson-gxl-s905l3b-B60av2.1.M.dtb"

setenv bootargs "root=/dev/mmcblk2p2 rootwait ro ${condev} fsck.repair=yes net.ifnames=0"

if fatload mmc 1 ${kernel_loadaddr} uImage && \
   fatload mmc 1 ${dtb_loadaddr} ${dtb_name}; then
    bootm ${kernel_loadaddr} - ${dtb_loadaddr}
fi
