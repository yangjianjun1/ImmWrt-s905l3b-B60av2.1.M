# S905L3B boot script - B60av2.1.M

setenv condev "console=ttyAML0,115200n8 no_console_suspend consoleblank=0"
setenv kernel_loadaddr "0x00208000"
setenv dtb_loadaddr "0x21800000"
setenv dtb_name "meson-gxl-s905l3b-b60-av21m.dtb"

setenv bootargs "root=LABEL=ROOTFS rootflags=data=writeback rw rootfstype=ext4 rootwait fsck.repair=yes net.ifnames=0 ${condev}"

# 尝试从 eMMC 启动
if fatload mmc 0 ${kernel_loadaddr} Image && \
   fatload mmc 0 ${dtb_loadaddr} ${dtb_name}; then
    echo "Booting from eMMC..."
    booti ${kernel_loadaddr} - ${dtb_loadaddr}
fi

# 尝试从 SD/TF 卡启动
if fatload mmc 1 ${kernel_loadaddr} Image && \
   fatload mmc 1 ${dtb_loadaddr} ${dtb_name}; then
    echo "Booting from SD card..."
    booti ${kernel_loadaddr} - ${dtb_loadaddr}
fi

# 尝试从 U 盘启动
usb start
if fatload usb 0 ${kernel_loadaddr} Image && \
   fatload usb 0 ${dtb_loadaddr} ${dtb_name}; then
    echo "Booting from USB..."
    booti ${kernel_loadaddr} - ${dtb_loadaddr}
fi
